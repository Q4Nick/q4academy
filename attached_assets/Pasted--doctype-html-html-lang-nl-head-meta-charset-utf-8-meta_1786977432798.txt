function el(name,attrs={}){
  const e=document.createElementNS(NS,name);
  for(const k in attrs) e.setAttribute(k,attrs[k]);
  return e;
}
function polar(r,deg){
  const a=deg*Math.PI/180;
  return [CX+r*Math.cos(a), CY+r*Math.sin(a)];
}
/* ringsegment tussen twee radii en twee hoeken */
function sector(r0,r1,a0,a1){
  const [x0,y0]=polar(r1,a0), [x1,y1]=polar(r1,a1);
  const [x2,y2]=polar(r0,a1), [x3,y3]=polar(r0,a0);
  const groot=(a1-a0)>180?1:0;
  return `M ${x0} ${y0} A ${r1} ${r1} 0 ${groot} 1 ${x1} ${y1} L ${x2} ${y2} A ${r0} ${r0} 0 ${groot} 0 ${x3} ${y3} Z`;
}
/* boog voor gebogen tekst; omgekeerd tekenen als het onderin staat */
function boog(r,a0,a1,omgekeerd){
  let s=a0,e=a1;
  if(omgekeerd){ s=a1; e=a0; }
  const [x0,y0]=polar(r,s), [x1,y1]=polar(r,e);
  const groot=Math.abs(e-s)>180?1:0;
  const sweep=e>s?1:0;
  return `M ${x0} ${y0} A ${r} ${r} 0 ${groot} ${sweep} ${x1} ${y1}`;
}
function normaliseer(deg){ let d=deg%360; if(d>=180)d-=360; if(d<-180)d+=360; return d; }
function moetOmgekeerd(midden){ return normaliseer(midden)>0; }

/* automatisch passende tekstgrootte binnen een boog */
function pasGrootte(tekst,r,spanGraden,max,min){
  const beschikbaar=(spanGraden*Math.PI/180)*r*0.90;
  const nodig=tekst.length*0.53;
  return Math.max(min,Math.min(max,beschikbaar/nodig));
}

/* kleurhulp: hex mengen met wit of zwart voor de verlopen */
function meng(hex,doel,t){
  const p=h=>[1,3,5].map(i=>parseInt(h.substr(i,2),16));
  const [r1,g1,b1]=p(hex), [r2,g2,b2]=p(doel);
  const m=(a,b)=>Math.round(a+(b-a)*t);
  return "#"+[m(r1,r2),m(g1,g2),m(b1,b2)].map(v=>v.toString(16).padStart(2,"0")).join("");
}
/* radiaal verloop: binnenrand iets dieper, buitenrand iets lichter — geeft de glans van het origineel */
function verloop(id,r,binnenKleur,buitenKleur,vanaf){
  const g=el("radialGradient",{id:id,gradientUnits:"userSpaceOnUse",cx:CX,cy:CY,r:r});
  g.appendChild(el("stop",{offset:vanaf,"stop-color":binnenKleur}));
  g.appendChild(el("stop",{offset:1,"stop-color":buitenKleur}));
  defs.appendChild(g);
  return "url(#"+id+")";
}

}

/* label in een subschaalband: één regel, of twee regels als het te klein zou worden */
function plaatsSubLabel(g,naam,band,a0,a1,kleur){
  const span=a1-a0;
  const stijl=fs=>({fill:kleur,"font-weight":700,"font-size":fs});
  const fs=pasGrootte(naam,band.mid,span,13,8);
  if(fs>=10.2 || naam.indexOf(" ")===-1){
    tekstOpBoog(g,naam,band.mid,a0,a1,{attrs:stijl(fs)});
    return;
  }
  const woorden=naam.split(" ");
  const helft=Math.ceil(woorden.length/2);
  const regel1=woorden.slice(0,helft).join(" ");
  const regel2=woorden.slice(helft).join(" ");
  const om=moetOmgekeerd((a0+a1)/2);
  const dr=6.5;
  const rEerste = om ? band.mid-dr : band.mid+dr;   // bovenste regel voor de kijker
  const rTweede = om ? band.mid+dr : band.mid-dr;
  const f=Math.min(
    pasGrootte(regel1,rEerste,span,11.5,8),
    pasGrootte(regel2,rTweede,span,11.5,8)
  );
  tekstOpBoog(g,regel1,rEerste,a0,a1,{attrs:stijl(f)});
  tekstOpBoog(g,regel2,rTweede,a0,a1,{attrs:stijl(f)});
}

    `<span class="naam">${comp.naam}</span>
     <span class="groep" style="color:${comp.kleur}">Composietschaal · ${comp.vraag}</span>
     <span class="uitleg">${comp.uitleg}</span>`;
  detailBox.innerHTML=subLijst(comp);
}
function toonHart(){
  selectieBox.style.borderLeftColor="#173b8f";
  selectieBox.innerHTML=
    `<span class="naam">${welbevinden.naam}</span>
     <span class="groep" style="color:#173b8f">${welbevinden.vraag}</span>
     <span class="uitleg">${welbevinden.uitleg}</span>`;
  detailBox.innerHTML="";
}
function leegPaneel(){
  selectieBox.style.borderLeftColor="#173b8f";
  selectieBox.innerHTML=
    `<span class="naam">Kies een composietschaal</span>
     <span class="uitleg">Klik op een van de vijf gekleurde segmenten in het wiel voor een korte uitleg, of op het hart voor Welbevinden.</span>`;
  detailBox.innerHTML="";
}

function kies(type,id){
  if(selectie && selectie.type===type && selectie.id===id){ reset(); return; }
  selectie={type,id};
  if(type==="comp"){ markeerComp(id); toonComp(id); }
  else { markeerHart(); toonHart(); }
  legendaBij();
}
function reset(){
  selectie=null;
  wisMarkering();
  leegPaneel();
  legendaBij();
}

/* Klikken op een subschaalband selecteert de hele composietschaal.
   Alleen de composietring en het hart zijn focusbaar met het toetsenbord. */
alleSegmenten.forEach(s=>{
  const type=s.dataset.type;
  const doelType = type==="hart" ? "hart" : "comp";
  const doelId   = type==="sub" ? s.dataset.comp : s.dataset.id;

  s.addEventListener("mouseenter",()=>{
    if(selectie) return;
    if(doelType==="comp") markeerComp(doelId); else markeerHart();
  });
  s.addEventListener("mouseleave",()=>{ if(!selectie) wisMarkering(); });
  s.addEventListener("click",()=>kies(doelType,doelId));
  if(s.hasAttribute("tabindex")){
    s.addEventListener("keydown",e=>{
      if(e.key==="Enter"||e.key===" "){ e.preventDefault(); kies(doelType,doelId); }
    });
  }
});

/* legenda */
const legend=document.getElementById("legend");
composieten.forEach(c=>{
  const li=document.createElement("li");
  const b=document.createElement("button");
  b.type="button";
  b.setAttribute("aria-pressed","false");
  b.dataset.comp=c.id;
  b.style.color=c.kleur;
  b.innerHTML=`<i class="dot" style="background:${c.kleur}"></i><span style="color:#34425d">${c.naam}</span>`;
  b.addEventListener("click",()=>kies("comp",c.id));
  li.appendChild(b); legend.appendChild(li);
});
function legendaBij(){
  legend.querySelectorAll("button").forEach(b=>{
    const aan = selectie && selectie.type==="comp" && selectie.id===b.dataset.comp;
    b.setAttribute("aria-pressed",String(!!aan));
  });
}
